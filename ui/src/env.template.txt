// controlPlaneMode is the front-door auth arm: 'iap' (deployed, behind
// Identity-Aware Proxy) or 'none' (local dev only). The data plane is always
// mediated — the backend brokers all data and media through /api signed URLs and
// the deny-all client rules are deployed — so there is no data-plane mode to
// choose or carry here. IAP authenticates the user and the backend stamps every
// request by that verified identity, so the client holds no session of its own.
interface AppEnv {
  controlPlaneMode: 'iap' | 'none';
}

export const env: AppEnv = {
  controlPlaneMode: '${UI_CONTROL_PLANE_MODE}',
};
